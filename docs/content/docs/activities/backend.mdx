**[`@robojs/server`](/docs/ref/@robojs/server)** provides file-based HTTP routing (your file names and folder structure automatically become endpoints) for your activity's server-side logic. Files in `/src/api` become API endpoints.

## Creating Routes

Each file in `src/api/` maps to an endpoint. The file name determines the route path. For example, `src/api/hello.ts` becomes `GET /api/hello`.

<Files>
  <Folder name="src" defaultOpen disabled>
    <Folder name="api" defaultOpen>
      <File name="hello.ts" note="GET /api/hello" />
      <File name="data.ts" note="POST /api/data" />
      <Folder name="users" defaultOpen>
        <File name="[id].ts" note="/api/users/:id" />
      </Folder>
    </Folder>
  </Folder>
</Files>

Return a string or object from the default export:

<Tabs items={["TypeScript", "JavaScript"]} groupId="language" persist>
<Tab value="TypeScript">

  ```ts title="src/api/hello.ts"
  export default () => {
    return { message: 'Hello, world!' }
  }
  ```

</Tab>
<Tab value="JavaScript">

  ```js title="src/api/hello.js"
  export default () => {
    return { message: 'Hello, world!' }
  }
  ```

</Tab>
</Tabs>

## Request and Response

Use `RoboRequest` to read query parameters, body, and headers. Return an object for JSON responses, or use `RoboResponse` for custom status codes and headers:

<Tabs items={["TypeScript", "JavaScript"]} groupId="language" persist>
<Tab value="TypeScript">

  ```ts title="src/api/data.ts"
  import type { RoboRequest } from '@robojs/server'

  export default async (req: RoboRequest) => {
    const body = await req.json()
    return { received: body }
  }
  ```

</Tab>
<Tab value="JavaScript">

  ```js title="src/api/data.js"
  export default async (req) => {
    const body = await req.json()
    return { received: body }
  }
  ```

</Tab>
</Tabs>

Access query parameters through `req.query` and URL parameters through `req.params`.

## The Token Endpoint

Every activity template includes `src/api/token.ts`, which exchanges an OAuth2 authorization code for an access token while keeping your client secret secure on the server. See [Authentication](./authentication) for the full implementation and details on the OAuth2 flow.

## Nested Routes

Create subdirectories for nested route paths. Dynamic segments use `[param]` syntax in the file name:

| File path | Route |
|-----------|-------|
| `src/api/users.ts` | `/api/users` |
| `src/api/users/[id].ts` | `/api/users/:id` |
| `src/api/guilds/[guildId]/members.ts` | `/api/guilds/:guildId/members` |

Access dynamic parameters through `req.params`:

<Tabs items={["TypeScript", "JavaScript"]} groupId="language" persist>
<Tab value="TypeScript">

  ```ts title="src/api/users/[id].ts"
  import type { RoboRequest } from '@robojs/server'

  export default (req: RoboRequest) => {
    const { id } = req.params
    return { userId: id }
  }
  ```

</Tab>
<Tab value="JavaScript">

  ```js title="src/api/users/[id].js"
  export default (req) => {
    const { id } = req.params
    return { userId: id }
  }
  ```

</Tab>
</Tabs>

## HTTP Methods

The default export handles all HTTP methods. Use named exports for method-specific handlers:

<Tabs items={["TypeScript", "JavaScript"]} groupId="language" persist>
<Tab value="TypeScript">

  ```ts title="src/api/users.ts"
  import type { RoboRequest } from '@robojs/server'

  export const GET = () => {
    return { users: [] }
  }

  export const POST = async (req: RoboRequest) => {
    const body = await req.json()
    return { created: body }
  }
  ```

</Tab>
<Tab value="JavaScript">

  ```js title="src/api/users.js"
  export const GET = () => {
    return { users: [] }
  }

  export const POST = async (req) => {
    const body = await req.json()
    return { created: body }
  }
  ```

</Tab>
</Tabs>

Supported named exports: `GET`, `POST`, `PUT`, `DELETE`, `PATCH`, `OPTIONS`, `HEAD`.

## Server Configuration

Configure the server plugin in `config/plugins/robojs/server.ts`:

<Tabs items={["TypeScript", "JavaScript"]} groupId="language" persist>
<Tab value="TypeScript">

  ```ts title="config/plugins/robojs/server.ts"
  export default {
    port: 3000,
    cors: true
  }
  ```

</Tab>
<Tab value="JavaScript">

  ```js title="config/plugins/robojs/server.js"
  export default {
    port: 3000,
    cors: true
  }
  ```

</Tab>
</Tabs>

| Option | Type | Default | Description |
|--------|------|---------|-------------|
| `port` | number | 3000 | Server port |
| `cors` | boolean | false | Enable CORS |
| `prefix` | string | `/api` | Route prefix |

The `PORT` environment variable overrides the `port` option.
